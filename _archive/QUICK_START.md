## 🧪 اختبر الآن:

```javascript
// 1. تحقق من تحميل الأنظمة
console.log(window.HSE_SYSTEMS);

// 2. اختبر رفع ملف
const testFile = new File(["test"], "test.txt");
HSE_SYSTEMS.FileUploadSystem.uploadFile(testFile)
  .then(r => console.log("✅ نجح:", r))
  .catch(e => console.error("❌ خطأ:", e));

// 3. اختبر إرسال إشعار
HSE_SYSTEMS.NotificationSystem.notifyUser(
  "test-user",
  "تجربة النظام",
  "الإشعارات تعمل بنجاح"
).then(() => console.log("✅ إشعار أرسل"));

// 4. اختبر إنشاء تصريح
HSE_SYSTEMS.PermitSystem.createPermit({
  title: "تصريح اختبار",
  approvers: ["approver-1"]
}).then(r => console.log("✅ تصريح أنشئ:", r.id))
 .catch(e => console.error("❌ خطأ:", e));
```
